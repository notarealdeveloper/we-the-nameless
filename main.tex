% Niqqud stripping Lua
\directlua{
  function strip_hebrew_niqqud(s)
    local out = {}

    for pos, code in utf8.codes(s) do
      if not (
        (code >= 0x0591 and code <= 0x05BD) or
        code == 0x05BF or
        (code >= 0x05C1 and code <= 0x05C2) or
        (code >= 0x05C4 and code <= 0x05C5) or
        code == 0x05C7
      ) then
        table.insert(out, utf8.char(code))
      end
    end

    return table.concat(out)
  end
}

\newcommand{\HebrewNoNiqqud}[1]{%
  \directlua{
    tex.sprint(strip_hebrew_niqqud("\luaescapestring{\detokenize{#1}}"))
  }%
}

% Hebrew -> Phoenician/Paleo-Hebrew Unicode
% This is for fonts that put Paleo/Phoenician glyphs at the
% Phoenician Unicode codepoints instead of the Hebrew codepoints.
%
% Important: do NOT use Lua -- line comments inside \directlua{...}.
% TeX turns most newlines in macro arguments into spaces, so a Lua
% line comment can accidentally comment out the rest of the chunk.
\directlua{
  hebrew_to_phoenician_map = {
    [0x05D0] = 0x10900,
    [0x05D1] = 0x10901,
    [0x05D2] = 0x10902,
    [0x05D3] = 0x10903,
    [0x05D4] = 0x10904,
    [0x05D5] = 0x10905,
    [0x05D6] = 0x10906,
    [0x05D7] = 0x10907,
    [0x05D8] = 0x10908,
    [0x05D9] = 0x10909,
    [0x05DA] = 0x1090A,
    [0x05DB] = 0x1090A,
    [0x05DC] = 0x1090B,
    [0x05DD] = 0x1090C,
    [0x05DE] = 0x1090C,
    [0x05DF] = 0x1090D,
    [0x05E0] = 0x1090D,
    [0x05E1] = 0x1090E,
    [0x05E2] = 0x1090F,
    [0x05E3] = 0x10910,
    [0x05E4] = 0x10910,
    [0x05E5] = 0x10911,
    [0x05E6] = 0x10911,
    [0x05E7] = 0x10912,
    [0x05E8] = 0x10913,
    [0x05E9] = 0x10914,
    [0x05EA] = 0x10915,
  }

  function hebrew_to_phoenician(s)
    s = strip_hebrew_niqqud(s)
    local out = {}

    for pos, code in utf8.codes(s) do
      local mapped = hebrew_to_phoenician_map[code]
      if mapped then
        table.insert(out, utf8.char(mapped))
      elseif code == 0x05BE then
        table.insert(out, " ")
      else
        table.insert(out, utf8.char(code))
      end
    end

    return table.concat(out)
  end
}

\newcommand{\HebrewToPhoenician}[1]{%
  \directlua{%
    tex.sprint(hebrew_to_phoenician("\luaescapestring{\detokenize{#1}}"))%
  }%
}

}

\def\englishsize{\normalsize}
\def\hebrewsize{\normalsize}

\newcommand{\VerseEnglish}[1]{%
    \begin{minipage}[t]{\VerseColWidth}
        \EnglishFont
        \englishsize
        \raggedright
        #1\par
    \end{minipage}%
}

\newcommand{\VerseHebrew}[1]{%
    \hbox to \VerseColWidth{%
        \hss
        \vtop{%
            \hsize=\VerseColWidth
            \begin{RTL}
            \HebrewFont
            \hebrewsize
            #1\par
            \end{RTL}
        }%
    }%
}

\newcommand{\VerseColumns}[2]{%
    \noindent
    \VerseEnglish{#1}%
    \hspace{\VerseGap}%
    \vrule width \VerseRuleWidth%
    \hspace{\VerseGap}%
    \VerseHebrew{#2}%
    \par
}

\newcommand{\VerseHeader}[1]{%
    \Needspace{6\baselineskip}
    \vspace{0.8em}%
    \hrule
    \vspace{0.25em}%
    {\centering \large \CurrentBook\ \CurrentChapter:#1\par}%
    \vspace{0.8em}%
    \hrule
    \vspace{0.3em}%
}

    \hfuzz=2em
    \VerseColumns{#3}{#2}%
    \endgroup
    \vspace{0.8em}%
    #4\par
    \vspace{0.8em}%
}
