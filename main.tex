% ------------------------------------------------------------
% Hebrew text encoders and font profiles
%
% Ground truth in 1-genesis/*.tex stays Hebrew.  A font profile decides
% how Hebrew input is encoded before it reaches a particular font.
%
% Examples:
%   \renewcommand{\hJProfile}{\HebrewProfileHebrewPaleoCodepointsJ}
%   \renewcommand{\hJProfile}{\HebrewProfilePaleoASCIIJ}
%   \renewcommand{\hJProfile}{\HebrewProfilePhoenicianUnicodeJ}
% ------------------------------------------------------------

\directlua{
  paleotex = paleotex or {}

  function paleotex.strip_hebrew_niqqud(s)
    local out = {}
    for pos, code in utf8.codes(s) do
      if not ((code >= 0x0591 and code <= 0x05BD) or code == 0x05BF or (code >= 0x05C1 and code <= 0x05C2) or (code >= 0x05C4 and code <= 0x05C5) or code == 0x05C7) then
        table.insert(out, utf8.char(code))
      end
    end
    return table.concat(out)
  end

  paleotex.hebrew_to_phoenician_unicode_map = {
    [0x05D0] = 0x10900, [0x05D1] = 0x10901, [0x05D2] = 0x10902, [0x05D3] = 0x10903,
    [0x05D4] = 0x10904, [0x05D5] = 0x10905, [0x05D6] = 0x10906, [0x05D7] = 0x10907,
    [0x05D8] = 0x10908, [0x05D9] = 0x10909, [0x05DA] = 0x1090A, [0x05DB] = 0x1090A,
    [0x05DC] = 0x1090B, [0x05DD] = 0x1090C, [0x05DE] = 0x1090C, [0x05DF] = 0x1090D,
    [0x05E0] = 0x1090D, [0x05E1] = 0x1090E, [0x05E2] = 0x1090F, [0x05E3] = 0x10910,
    [0x05E4] = 0x10910, [0x05E5] = 0x10911, [0x05E6] = 0x10911, [0x05E7] = 0x10912,
    [0x05E8] = 0x10913, [0x05E9] = 0x10914, [0x05EA] = 0x10915
  }

  paleotex.hebrew_to_paleo_ascii_map = {
    [0x05D0] = "A", [0x05D1] = "B", [0x05D2] = "G", [0x05D3] = "D",
    [0x05D4] = "H", [0x05D5] = "W", [0x05D6] = "Z", [0x05D7] = "X",
    [0x05D8] = "J", [0x05D9] = "Y", [0x05DA] = "K", [0x05DB] = "K",
    [0x05DC] = "L", [0x05DD] = "M", [0x05DE] = "M", [0x05DF] = "N",
    [0x05E0] = "N", [0x05E1] = "S", [0x05E2] = "]", [0x05E3] = "P",
    [0x05E4] = "P", [0x05E5] = "C", [0x05E6] = "C", [0x05E7] = "Q",
    [0x05E8] = "R", [0x05E9] = "V", [0x05EA] = "T"
  }

  function paleotex.map_hebrew(s, map, unicode_output)
    s = paleotex.strip_hebrew_niqqud(s)
    local out = {}
    for pos, code in utf8.codes(s) do
      local mapped = map[code]
      if mapped then
        if unicode_output then
          table.insert(out, utf8.char(mapped))
        else
          table.insert(out, mapped)
        end
      elseif code == 0x05BE then
        table.insert(out, " ")
      else
        table.insert(out, utf8.char(code))
      end
    end
    return table.concat(out)
  end

  function strip_hebrew_niqqud(s)
    return paleotex.strip_hebrew_niqqud(s)
  end

  function hebrew_to_phoenician(s)
    return paleotex.map_hebrew(s, paleotex.hebrew_to_phoenician_unicode_map, true)
  end

  function hebrew_to_paleo_ascii(s)
    return paleotex.map_hebrew(s, paleotex.hebrew_to_paleo_ascii_map, false)
  end
}

\newcommand{\HebrewNoNiqqud}[1]{%
  \directlua{tex.sprint(strip_hebrew_niqqud("\luaescapestring{\detokenize{#1}}"))}%
}

\newcommand{\HebrewToPhoenician}[1]{%
  \directlua{tex.sprint(hebrew_to_phoenician("\luaescapestring{\detokenize{#1}}"))}%
}

\newcommand{\HebrewToPaleoASCII}[1]{%
  \directlua{tex.sprint(hebrew_to_paleo_ascii("\luaescapestring{\detokenize{#1}}"))}%
}

    \par

    % Discourage page breaks within the block.
    \interlinepenalty=#2\relax
    \predisplaypenalty=#2\relax
    \postdisplaypenalty=#2\relax
    \interdisplaylinepenalty=#2\relax
    \@beginparpenalty=#2\relax

}
{
