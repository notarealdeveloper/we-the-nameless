\aB{What's this all about?}

\aA{

\begin{quote}
	
	\textbf{\hAJ{קין}} (\textit{q-y-n / qayin}) \emph{n.}\\
	Name of \emph{Cain}, meaning \emph{maker}, as in \emph{smith}, or \emph{metalworker}.\\
	Phonetically linked to \hAJ{קנה} (\textit{q-n-h / qānāh}), meaning\\
	\emph{to acquire}, \emph{to get}, \emph{to make}, \emph{to create}.
	
	\medskip
	
	\textbf{\hAJ{את}} (\textit{ʾet}) \emph{part.}\\
	(1) \emph{Direct object marker} (untranslated): marks a definite object.\\
	(2) \emph{Preposition} \emph{“with”}: expresses accompaniment or association.

	\medskip

	\textbf{\hAJ{ידע}} (\textit{y-d-ʿ / yādaʿ}) \emph{v.}\\
	Meaning \emph{to know}; in context, implies \emph{intimate} or \emph{sexual knowledge}.%
		\aC{\footnote{\footnotesize
			Translator's Note: A common feature of every known human language is that the most basic words
			in the language possess, among their possible meanings, more sexual connotations than their
			more complex verbal counterparts, e.g., compare:
			\begin{itemize}
			\item the verbs ``come'', ``do'', ``take'', ``need'', ``get (some)'', on the one hand,\\
			versus ``arrive'', ``carry out'', ``acquire'', ``require'', ``obtain (some)'' on the other;
			\item the adjectives ``hard'', ``open'', ``spread'', ``hot'', ``thirsty'', ``dirty'', ``wet'', on the one hand,\\
			versus ``inflexible'', ``accessible'', ``disperse'', ``scorching'', ``boiling'', ``unwashed'', ``damp'' on the other;
			\item the nouns ``thing'', ``part'', ``member'', ``body'', ``bed'', ``head'', ``rear'', ``top'', on the one hand,\\
			versus ``object'', ``portion'', ``participant'', ``form'', ``mattress'', ``cranium'', ``stern'', ``summit''
			on the other.
			\end{itemize}			 
			The attentive reader may object that these are not, in fact, synonyms.
			We agree, and that is precisely the point.
		}
	}
	
	\end{quote}

}
