\Chapter{8}

\Verse{1}
{
	\hP{וַיִּזְכֹּר אֱלֹהִים אֶת נֹחַ וְאֵת כּל הַחַיָּה וְאֶת כּל הַבְּהֵמָה אֲשֶׁר אִתּוֹ בַּתֵּבָה וַיַּעֲבֵר אֱלֹהִים רוּחַ עַל הָאָרֶץ וַיָּשֹׁכּוּ הַמָּיִם}
}
{
	\eP{
		And God remembered Noah\\
		and all the wild animals\\
		and all the domestic animals that were with him in the ark,\\

		and the water decreased.\\
	}
}
{
}

}

\Verse{3}

	\hP{וַיַּחְסְרוּ הַמַּיִם מִקְצֵה חֲמִשִּׁים וּמְאַת יוֹם}

\Verse{4}
{

}
{
	\eP{

\Verse{5}

	\hP{וְהַמַּיִם הָיוּ הָלוֹךְ וְחָסוֹר עַד הַחֹדֶשׁ הָעֲשִׂירִי בָּעֲשִׂירִי בְּאֶחָד לַחֹדֶשׁ נִרְאוּ רָאשֵׁי הֶהָרִים}
}
{
	\eP{
		And the water went on receding until the tenth month.\\

}
{
}

\Verse{6}
{
	\hJ{וַיְהִי מִקֵּץ אַרְבָּעִים יוֹם וַיִּפְתַּח נֹחַ אֶת חַלּוֹן הַתֵּבָה אֲשֶׁר עָשָׂה}
}
{
	\eJ{

}

\Verse{7}
{
	\hP{וַיְשַׁלַּח אֶת הָעֹרֵב וַיֵּצֵא יָצוֹא וָשׁוֹב עַד יְבֹשֶׁת הַמַּיִם מֵעַל הָאָרֶץ}
}
{
	\eP{
		And he let a raven go,\\

	}
}
{
}

\Verse{8}
{
	\hJ{וַיְשַׁלַּח אֶת הַיּוֹנָה מֵאִתּוֹ לִרְאוֹת הֲקַלּוּ הַמַּיִם מֵעַל פְּנֵי הָאֲדָמָה}
}
{
	\eJ{
		And he let a dove go from him\\

	}
}
{
}

\Verse{9}
{
	\hJ{וְלֹא מָצְאָה הַיּוֹנָה מָנוֹחַ לְכַף רַגְלָהּ וַתָּשׁב אֵלָיו אֶל הַתֵּבָה כִּי מַיִם עַל פְּנֵי כל הָאָרֶץ וַיִּשְׁלַח יָדוֹ וַיִּקָּחֶהָ וַיָּבֵא אֹתָהּ אֵלָיו אֶל הַתֵּבָה}
}
{
	\eJ{
		And the dove did not find a resting place for its foot,\\
		and it came back to him to the ark,\\
		for waters were on the face of the earth,\\
		and he put out his hand and took it\\
		and brought it to him to the ark.\\
	}
}
{
}

\Verse{10}
{
	\hJ{וַיָּחֶל עוֹד שִׁבְעַת יָמִים אֲחֵרִים וַיֹּסֶף שַׁלַּח אֶת הַיּוֹנָה מִן הַתֵּבָה}

}
{
	\eJ{

		and be fruitful and multiply on the earth.”\\

\Verse{21}

		and I won’t strike all the living again as I have done.\\
	}
}
{
}
