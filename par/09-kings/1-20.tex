}
{
}

	\eX{

}
{
}

\Verse{10}

}
{
	\eX{

}
{
}

}
{
	\eX{

}

	\eX{

{
}

	\eX{

{
}

		“Thus said YHWH: Because the Arameans have said, ‘YHWH is a God of

	}
}
{

{
