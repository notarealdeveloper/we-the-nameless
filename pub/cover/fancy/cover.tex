\definecolor{SpineForeground}{cmyk}{0.00,0.20,0.62,0.30}

\newlength{\CoverNamelessY}
\setlength{\CoverNamelessY}{4.725in}
\newlength{\CoverVolumeY}
\setlength{\CoverVolumeY}{1.425in}
\newcommand{\SpineLabel}[2]{%
  \adjustbox{max width=\dimexpr\CoverSpineWidth-0.125in\relax}{\fontsize{#1}{#1}\selectfont #2}%
}
\input{ld-logo.tex}
\def\CoverSpineLD{ld}
\newcommand{\SpineName}{%
  \ifx\CoverSpine\CoverSpineLD
    % Keep the mark about as tall as the front-cover "The Nameless", but use
    % the taller 50:64 proportions requested for the spine.
    \SpineLabel{22}{\LD[1em][0.78em]}%
  \else
    \SpineLabel{15}{\shortstack{The\\Nameless}}%
  \fi
}

\ifdefined\CoverShowSpine\else\def\CoverShowSpine{true}\fi
\def\CoverShowSpineTrue{true}
\begin{document}
\begin{tikzpicture}[remember picture,overlay,x=1in,y=1in]
  % Each 2:3 source is scaled to panel width and clipped vertically, preserving
  % its geometry while filling a 7.125 x 10.25 in panel through the outer bleed.
  \begin{scope}
    \clip (current page.south west) rectangle ([xshift=7.125in]current page.north west);
    \node[anchor=center,inner sep=0pt] at ([xshift=3.5625in]current page.west)
      {\includegraphics[width=7.125in]{\CoverProjectDir/assets/back.png}};
  \end{scope}

  \begin{scope}
    \clip ([xshift=\dimexpr 7.125in+\CoverSpineWidth\relax]current page.south west)
      rectangle (current page.north east);
    \node[anchor=center,inner sep=0pt] at ([xshift=-3.5625in]current page.east)
      {\includegraphics[width=7.125in]{\CoverProjectDir/assets/back.png}};
    % Use the text-free artwork so all three titles can be positioned exactly.
    \node[SpineForeground] at
      ([xshift=-\CoverFrontCenterX,yshift=\CoverWeY]current page.south east)
      {\fontsize{100}{120}\selectfont\textsc{We}};
    \node[SpineForeground] at
      ([xshift=-\CoverFrontCenterX,yshift=\CoverNamelessY]current page.south east)
      {\fontsize{25}{30}\selectfont The Nameless};
    \node[SpineForeground] at
      ([xshift=-\CoverFrontCenterX,yshift=\CoverVolumeY]current page.south east)
      {\fontsize{23}{28}\selectfont \CoverVolume};
  \end{scope}

  % Continue the celestial engraving onto the spine at the same scale as
  % the panels. Clip its central strip rather than squeezing the artwork.
  \begin{scope}
    \clip ([xshift=7.125in]current page.south west)
      rectangle ([xshift=\dimexpr 7.125in+\CoverSpineWidth\relax]current page.north west);
    \node[anchor=center,inner sep=0pt] at
      ([xshift=\dimexpr 7.125in+\CoverSpineWidth/2\relax]current page.west)
      {\includegraphics[width=7.125in]{\CoverProjectDir/assets/back.png}};
    \foreach \edge in {7.165in,\dimexpr 7.085in+\CoverSpineWidth\relax} {
      \draw[SpineForeground,line width=0.3pt]
        ([xshift=\edge]current page.south west) --
        ([xshift=\edge]current page.north west);
    }
  \end{scope}
  \ifx\CoverShowSpine\CoverShowSpineTrue
  \node[SpineForeground] at
    ([xshift=\dimexpr 7.125in+\CoverSpineWidth/2\relax,yshift=\CoverWeY]current page.south west)
    {\SpineLabel{26}{\textsc{We}}};
  \node[SpineForeground,align=center] at
    ([xshift=\dimexpr 7.125in+\CoverSpineWidth/2\relax,yshift=\CoverNamelessY]current page.south west)
    {\SpineName};
  \node[SpineForeground] at
    ([xshift=\dimexpr 7.125in+\CoverSpineWidth/2\relax,yshift=\CoverVolumeY]current page.south west)
    {\SpineLabel{18}{\CoverVolume}};
  \fi
\end{tikzpicture}
