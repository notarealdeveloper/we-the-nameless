% Shared geometric LD mark.  Keep the artwork independent of font metrics and
% leave its color unset so it inherits the surrounding text/TikZ color.
\providecommand{\LD}[1][1em]{%
  \tikz[baseline=0pt,x=#1,y=#1]{%
    % The nominal 1 x 1 box includes a narrow optical margin.  Each component
    % is the same square-cornered L, with the second rotated by 180 degrees.
    \path[use as bounding box] (0,0) rectangle (1,1);
    \fill
      (0.04,0.04) rectangle (0.18,0.69)
      (0.04,0.04) rectangle (0.69,0.18)
      (0.82,0.31) rectangle (0.96,0.96)
      (0.31,0.82) rectangle (0.96,0.96);
  }%
}
