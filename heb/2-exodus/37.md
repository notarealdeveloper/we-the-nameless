Exodus 37:1	וַיַּעַשׂ בְּצַלְאֵל אֶת־הָאָרֹן עֲצֵי שִׁטִּים אַמָּתַיִם וָחֵצִי ארְכּוֹ וְאַמָּה וָחֵצִי רחְבּוֹ וְאַמָּה וָחֵצִי קֹמָתוֹ׃
Exodus 37:2	וַיְצַפֵּהוּ זָהָב טָהוֹר מִבַּיִת וּמִחוּץ וַיַּעַשׂ לוֹ זֵר זָהָב סָבִיב׃
Exodus 37:3	וַיִּצֹק לוֹ אַרְבַּע טַבְּעֹת זָהָב עַל אַרְבַּע פַּעֲמֹתָיו וּשְׁתֵּי טַבָּעֹת עַל־צַלְעוֹ הָאֶחָת וּשְׁתֵּי טַבָּעֹת עַל־צַלְעוֹ הַשֵּׁנִית׃
Exodus 37:4	וַיַּעַשׂ בַּדֵּי עֲצֵי שִׁטִּים וַיְצַף אֹתָם זָהָב׃
Exodus 37:5	וַיָּבֵא אֶת־הַבַּדִּים בַּטַּבָּעֹת עַל צַלְעֹת הָאָרֹן לָשֵׂאת אֶת־הָאָרֹן׃
Exodus 37:6	וַיַּעַשׂ כַּפֹּרֶת זָהָב טָהוֹר אַמָּתַיִם וָחֵצִי ארְכָּהּ וְאַמָּה וָחֵצִי רחְבָּהּ׃
Exodus 37:7	וַיַּעַשׂ שְׁנֵי כְרֻבִים זָהָב מִקְשָׁה עָשָׂה אֹתָם מִשְּׁנֵי קְצוֹת הַכַּפֹּרֶת׃
Exodus 37:8	כְּרוּב־אֶחָד מִקָּצָה מִזֶּה וּכְרוּב־אֶחָד מִקָּצָה מִזֶּה מִן־הַכַּפֹּרֶת עָשָׂה אֶת־הַכְּרֻבִים מִשְּׁנֵי (קצוותו) [קְצוֹתָיו]׃
Exodus 37:9	וַיִּהְיוּ הַכְּרֻבִים פֹּרְשֵׂי כְנָפַיִם לְמַעְלָה סֹכְכִים בְּכַנְפֵיהֶם עַל־הַכַּפֹּרֶת וּפְנֵיהֶם אִישׁ אֶל־אָחִיו אֶל־הַכַּפֹּרֶת הָיוּ פְּנֵי הַכְּרֻבִים׃ {פ}
Exodus 37:10	וַיַּעַשׂ אֶת־הַשֻּׁלְחָן עֲצֵי שִׁטִּים אַמָּתַיִם ארְכּוֹ וְאַמָּה רחְבּוֹ וְאַמָּה וָחֵצִי קֹמָתוֹ׃
Exodus 37:11	וַיְצַף אֹתוֹ זָהָב טָהוֹר וַיַּעַשׂ לוֹ זֵר זָהָב סָבִיב׃
Exodus 37:12	וַיַּעַשׂ לוֹ מִסְגֶּרֶת טֹפַח סָבִיב וַיַּעַשׂ זֵר־זָהָב לְמִסְגַּרְתּוֹ סָבִיב׃
Exodus 37:13	וַיִּצֹק לוֹ אַרְבַּע טַבְּעֹת זָהָב וַיִּתֵּן אֶת־הַטַּבָּעֹת עַל אַרְבַּע הַפֵּאֹת אֲשֶׁר לְאַרְבַּע רַגְלָיו׃
Exodus 37:14	לְעֻמַּת הַמִּסְגֶּרֶת הָיוּ הַטַּבָּעֹת בָּתִּים לַבַּדִּים לָשֵׂאת אֶת־הַשֻּׁלְחָן׃
Exodus 37:15	וַיַּעַשׂ אֶת־הַבַּדִּים עֲצֵי שִׁטִּים וַיְצַף אֹתָם זָהָב לָשֵׂאת אֶת־הַשֻּׁלְחָן׃
Exodus 37:16	וַיַּעַשׂ אֶת־הַכֵּלִים  אֲשֶׁר עַל־הַשֻּׁלְחָן אֶת־קְעָרֹתָיו וְאֶת־כַּפֹּתָיו וְאֵת מְנַקִּיֹּתָיו וְאֶת־הַקְּשָׂוֺת אֲשֶׁר יֻסַּךְ בָּהֵן זָהָב טָהוֹר׃ {פ}
Exodus 37:17	וַיַּעַשׂ אֶת־הַמְּנֹרָה זָהָב טָהוֹר מִקְשָׁה עָשָׂה אֶת־הַמְּנֹרָה יְרֵכָהּ וְקָנָהּ גְּבִיעֶיהָ כַּפְתֹּרֶיהָ וּפְרָחֶיהָ מִמֶּנָּה הָיוּ׃
Exodus 37:18	וְשִׁשָּׁה קָנִים יֹצְאִים מִצִּדֶּיהָ שְׁלֹשָׁה  קְנֵי מְנֹרָה מִצִּדָּהּ הָאֶחָד וּשְׁלֹשָׁה קְנֵי מְנֹרָה מִצִּדָּהּ הַשֵּׁנִי׃
Exodus 37:19	שְׁלֹשָׁה גְבִעִים מְשֻׁקָּדִים בַּקָּנֶה הָאֶחָד כַּפְתֹּר וָפֶרַח וּשְׁלֹשָׁה גְבִעִים מְשֻׁקָּדִים בְּקָנֶה אֶחָד כַּפְתֹּר וָפָרַח כֵּן לְשֵׁשֶׁת הַקָּנִים הַיֹּצְאִים מִן־הַמְּנֹרָה׃
Exodus 37:20	וּבַמְּנֹרָה אַרְבָּעָה גְבִעִים מְשֻׁקָּדִים כַּפְתֹּרֶיהָ וּפְרָחֶיהָ׃
Exodus 37:21	וְכַפְתֹּר תַּחַת שְׁנֵי הַקָּנִים מִמֶּנָּה וְכַפְתֹּר תַּחַת שְׁנֵי הַקָּנִים מִמֶּנָּה וְכַפְתֹּר תַּחַת־שְׁנֵי הַקָּנִים מִמֶּנָּה לְשֵׁשֶׁת הַקָּנִים הַיֹּצְאִים מִמֶּנָּה׃
Exodus 37:22	כַּפְתֹּרֵיהֶם וּקְנֹתָם מִמֶּנָּה הָיוּ כֻּלָּהּ מִקְשָׁה אַחַת זָהָב טָהוֹר׃
Exodus 37:23	וַיַּעַשׂ אֶת־נֵרֹתֶיהָ שִׁבְעָה וּמַלְקָחֶיהָ וּמַחְתֹּתֶיהָ זָהָב טָהוֹר׃
Exodus 37:24	כִּכָּר זָהָב טָהוֹר עָשָׂה אֹתָהּ וְאֵת כּל־כֵּלֶיהָ׃ {פ}
Exodus 37:25	וַיַּעַשׂ אֶת־מִזְבַּח הַקְּטֹרֶת עֲצֵי שִׁטִּים אַמָּה ארְכּוֹ וְאַמָּה רחְבּוֹ רָבוּעַ וְאַמָּתַיִם קֹמָתוֹ מִמֶּנּוּ הָיוּ קַרְנֹתָיו׃
Exodus 37:26	וַיְצַף אֹתוֹ זָהָב טָהוֹר אֶת־גַּגּוֹ וְאֶת־קִירֹתָיו סָבִיב וְאֶת־קַרְנֹתָיו וַיַּעַשׂ לוֹ זֵר זָהָב סָבִיב׃
Exodus 37:27	וּשְׁתֵּי טַבְּעֹת זָהָב עָשָׂה־לוֹ  מִתַּחַת לְזֵרוֹ עַל שְׁתֵּי צַלְעֹתָיו עַל שְׁנֵי צִדָּיו לְבָתִּים לְבַדִּים לָשֵׂאת אֹתוֹ בָּהֶם׃
Exodus 37:28	וַיַּעַשׂ אֶת־הַבַּדִּים עֲצֵי שִׁטִּים וַיְצַף אֹתָם זָהָב׃
Exodus 37:29	וַיַּעַשׂ אֶת־שֶׁמֶן הַמִּשְׁחָה קֹדֶשׁ וְאֶת־קְטֹרֶת הַסַּמִּים טָהוֹר מַעֲשֵׂה רֹקֵחַ׃ {ס}
