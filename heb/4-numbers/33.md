Numbers 33:1	אֵלֶּה מַסְעֵי בְנֵי־יִשְׂרָאֵל אֲשֶׁר יָצְאוּ מֵאֶרֶץ מִצְרַיִם לְצִבְאֹתָם בְּיַד־מֹשֶׁה וְאַהֲרֹן׃
Numbers 33:2	וַיִּכְתֹּב מֹשֶׁה אֶת־מוֹצָאֵיהֶם לְמַסְעֵיהֶם עַל־פִּי יְהֹוָה וְאֵלֶּה מַסְעֵיהֶם לְמוֹצָאֵיהֶם׃
Numbers 33:3	וַיִּסְעוּ מֵרַעְמְסֵס בַּחֹדֶשׁ הָרִאשׁוֹן בַּחֲמִשָּׁה עָשָׂר יוֹם לַחֹדֶשׁ הָרִאשׁוֹן מִמּחֳרַת הַפֶּסַח יָצְאוּ בְנֵי־יִשְׂרָאֵל בְּיָד רָמָה לְעֵינֵי כּל־מִצְרָיִם׃
Numbers 33:4	וּמִצְרַיִם מְקַבְּרִים אֵת אֲשֶׁר הִכָּה יְהֹוָה בָּהֶם כּל־בְּכוֹר וּבֵאלֹהֵיהֶם עָשָׂה יְהֹוָה שְׁפָטִים׃
Numbers 33:5	וַיִּסְעוּ בְנֵי־יִשְׂרָאֵל מֵרַעְמְסֵס וַיַּחֲנוּ בְּסֻכֹּת׃
Numbers 33:6	וַיִּסְעוּ מִסֻּכֹּת וַיַּחֲנוּ בְאֵתָם אֲשֶׁר בִּקְצֵה הַמִּדְבָּר׃
Numbers 33:7	וַיִּסְעוּ מֵאֵתָם וַיָּשׁב עַל־פִּי הַחִירֹת אֲשֶׁר עַל־פְּנֵי בַּעַל צְפוֹן וַיַּחֲנוּ לִפְנֵי מִגְדֹּל׃
Numbers 33:8	וַיִּסְעוּ מִפְּנֵי הַחִירֹת וַיַּעַבְרוּ בְתוֹךְ־הַיָּם הַמִּדְבָּרָה וַיֵּלְכוּ דֶּרֶךְ שְׁלֹשֶׁת יָמִים בְּמִדְבַּר אֵתָם וַיַּחֲנוּ בְּמָרָה׃
Numbers 33:9	וַיִּסְעוּ מִמָּרָה וַיָּבֹאוּ אֵילִמָה וּבְאֵילִם שְׁתֵּים עֶשְׂרֵה עֵינֹת מַיִם וְשִׁבְעִים תְּמָרִים וַיַּחֲנוּ־שָׁם׃
Numbers 33:10	וַיִּסְעוּ מֵאֵילִם וַיַּחֲנוּ עַל־יַם־סוּף׃
Numbers 33:11	וַיִּסְעוּ מִיַּם־סוּף וַיַּחֲנוּ בְּמִדְבַּר־סִין׃
Numbers 33:12	וַיִּסְעוּ מִמִּדְבַּר־סִין וַיַּחֲנוּ בְּדפְקָה׃
Numbers 33:13	וַיִּסְעוּ מִדּפְקָה וַיַּחֲנוּ בְּאָלוּשׁ׃
Numbers 33:14	וַיִּסְעוּ מֵאָלוּשׁ וַיַּחֲנוּ בִּרְפִידִם וְלֹא־הָיָה שָׁם מַיִם לָעָם לִשְׁתּוֹת׃
Numbers 33:15	וַיִּסְעוּ מֵרְפִידִם וַיַּחֲנוּ בְּמִדְבַּר סִינָי׃
Numbers 33:16	וַיִּסְעוּ מִמִּדְבַּר סִינָי וַיַּחֲנוּ בְּקִבְרֹת הַתַּאֲוָה׃
Numbers 33:17	וַיִּסְעוּ מִקִּבְרֹת הַתַּאֲוָה וַיַּחֲנוּ בַּחֲצֵרֹת׃
Numbers 33:18	וַיִּסְעוּ מֵחֲצֵרֹת וַיַּחֲנוּ בְּרִתְמָה׃
Numbers 33:19	וַיִּסְעוּ מֵרִתְמָה וַיַּחֲנוּ בְּרִמֹּן פָּרֶץ׃
Numbers 33:20	וַיִּסְעוּ מֵרִמֹּן פָּרֶץ וַיַּחֲנוּ בְּלִבְנָה׃
Numbers 33:21	וַיִּסְעוּ מִלִּבְנָה וַיַּחֲנוּ בְּרִסָּה׃
Numbers 33:22	וַיִּסְעוּ מֵרִסָּה וַיַּחֲנוּ בִּקְהֵלָתָה׃
Numbers 33:23	וַיִּסְעוּ מִקְּהֵלָתָה וַיַּחֲנוּ בְּהַר־שָׁפֶר׃
Numbers 33:24	וַיִּסְעוּ מֵהַר־שָׁפֶר וַיַּחֲנוּ בַּחֲרָדָה׃
Numbers 33:25	וַיִּסְעוּ מֵחֲרָדָה וַיַּחֲנוּ בְּמַקְהֵלֹת׃
Numbers 33:26	וַיִּסְעוּ מִמַּקְהֵלֹת וַיַּחֲנוּ בְּתָחַת׃
Numbers 33:27	וַיִּסְעוּ מִתָּחַת וַיַּחֲנוּ בְּתָרַח׃
Numbers 33:28	וַיִּסְעוּ מִתָּרַח וַיַּחֲנוּ בְּמִתְקָה׃
Numbers 33:29	וַיִּסְעוּ מִמִּתְקָה וַיַּחֲנוּ בְּחַשְׁמֹנָה׃
Numbers 33:30	וַיִּסְעוּ מֵחַשְׁמֹנָה וַיַּחֲנוּ בְּמֹסֵרוֹת׃
Numbers 33:31	וַיִּסְעוּ מִמֹּסֵרוֹת וַיַּחֲנוּ בִּבְנֵי יַעֲקָן׃
Numbers 33:32	וַיִּסְעוּ מִבְּנֵי יַעֲקָן וַיַּחֲנוּ בְּחֹר הַגִּדְגָּד׃
Numbers 33:33	וַיִּסְעוּ מֵחֹר הַגִּדְגָּד וַיַּחֲנוּ בְּיטְבָתָה׃
Numbers 33:34	וַיִּסְעוּ מִיּטְבָתָה וַיַּחֲנוּ בְּעַבְרֹנָה׃
Numbers 33:35	וַיִּסְעוּ מֵעַבְרֹנָה וַיַּחֲנוּ בְּעֶצְיֹן גָּבֶר׃
Numbers 33:36	וַיִּסְעוּ מֵעֶצְיֹן גָּבֶר וַיַּחֲנוּ בְמִדְבַּר־צִן הִוא קָדֵשׁ׃
Numbers 33:37	וַיִּסְעוּ מִקָּדֵשׁ וַיַּחֲנוּ בְּהֹר הָהָר בִּקְצֵה אֶרֶץ אֱדוֹם׃
Numbers 33:38	וַיַּעַל אַהֲרֹן הַכֹּהֵן אֶל־הֹר הָהָר עַל־פִּי יְהֹוָה וַיָּמת שָׁם בִּשְׁנַת הָאַרְבָּעִים לְצֵאת בְּנֵי־יִשְׂרָאֵל מֵאֶרֶץ מִצְרַיִם בַּחֹדֶשׁ הַחֲמִישִׁי בְּאֶחָד לַחֹדֶשׁ׃
Numbers 33:39	וְאַהֲרֹן בֶּן־שָׁלֹשׁ וְעֶשְׂרִים וּמְאַת שָׁנָה בְּמֹתוֹ בְּהֹר הָהָר׃ {ס}        
Numbers 33:40	וַיִּשְׁמַע הַכְּנַעֲנִי מֶלֶךְ עֲרָד וְהוּא־יֹשֵׁב בַּנֶּגֶב בְּאֶרֶץ כְּנָעַן בְּבֹא בְּנֵי יִשְׂרָאֵל׃
Numbers 33:41	וַיִּסְעוּ מֵהֹר הָהָר וַיַּחֲנוּ בְּצַלְמֹנָה׃
Numbers 33:42	וַיִּסְעוּ מִצַּלְמֹנָה וַיַּחֲנוּ בְּפוּנֹן׃
Numbers 33:43	וַיִּסְעוּ מִפּוּנֹן וַיַּחֲנוּ בְּאֹבֹת׃
Numbers 33:44	וַיִּסְעוּ מֵאֹבֹת וַיַּחֲנוּ בְּעִיֵּי הָעֲבָרִים בִּגְבוּל מוֹאָב׃
Numbers 33:45	וַיִּסְעוּ מֵעִיִּים וַיַּחֲנוּ בְּדִיבֹן גָּד׃
Numbers 33:46	וַיִּסְעוּ מִדִּיבֹן גָּד וַיַּחֲנוּ בְּעַלְמֹן דִּבְלָתָיְמָה׃
Numbers 33:47	וַיִּסְעוּ מֵעַלְמֹן דִּבְלָתָיְמָה וַיַּחֲנוּ בְּהָרֵי הָעֲבָרִים לִפְנֵי נְבוֹ׃
Numbers 33:48	וַיִּסְעוּ מֵהָרֵי הָעֲבָרִים וַיַּחֲנוּ בְּעַרְבֹת מוֹאָב עַל יַרְדֵּן יְרֵחוֹ׃
Numbers 33:49	וַיַּחֲנוּ עַל־הַיַּרְדֵּן מִבֵּית הַיְשִׁמֹת עַד אָבֵל הַשִּׁטִּים בְּעַרְבֹת מוֹאָב׃ {ס}        
Numbers 33:50	וַיְדַבֵּר יְהֹוָה אֶל־מֹשֶׁה בְּעַרְבֹת מוֹאָב עַל־יַרְדֵּן יְרֵחוֹ לֵאמֹר׃
Numbers 33:51	דַּבֵּר אֶל־בְּנֵי יִשְׂרָאֵל וְאָמַרְתָּ אֲלֵהֶם כִּי אַתֶּם עֹבְרִים אֶת־הַיַּרְדֵּן אֶל־אֶרֶץ כְּנָעַן׃
Numbers 33:52	וְהוֹרַשְׁתֶּם אֶת־כּל־יֹשְׁבֵי הָאָרֶץ מִפְּנֵיכֶם וְאִבַּדְתֶּם אֵת כּל־מַשְׂכִּיֹּתָם וְאֵת כּל־צַלְמֵי מַסֵּכֹתָם תְּאַבֵּדוּ וְאֵת כּל־בָּמוֹתָם תַּשְׁמִידוּ׃
Numbers 33:53	וְהוֹרַשְׁתֶּם אֶת־הָאָרֶץ וִישַׁבְתֶּם־בָּהּ כִּי לָכֶם נָתַתִּי אֶת־הָאָרֶץ לָרֶשֶׁת אֹתָהּ׃
Numbers 33:54	וְהִתְנַחַלְתֶּם אֶת־הָאָרֶץ בְּגוֹרָל לְמִשְׁפְּחֹתֵיכֶם לָרַב תַּרְבּוּ אֶת־נַחֲלָתוֹ וְלַמְעַט תַּמְעִיט אֶת־נַחֲלָתוֹ אֶל אֲשֶׁר־יֵצֵא לוֹ שָׁמָּה הַגּוֹרָל לוֹ יִהְיֶה לְמַטּוֹת אֲבֹתֵיכֶם תִּתְנֶחָלוּ׃
Numbers 33:55	וְאִם־לֹא תוֹרִישׁוּ אֶת־יֹשְׁבֵי הָאָרֶץ מִפְּנֵיכֶם וְהָיָה אֲשֶׁר תּוֹתִירוּ מֵהֶם לְשִׂכִּים בְּעֵינֵיכֶם וְלִצְנִינִם בְּצִדֵּיכֶם וְצָרְרוּ אֶתְכֶם עַל־הָאָרֶץ אֲשֶׁר אַתֶּם יֹשְׁבִים בָּהּ׃
Numbers 33:56	וְהָיָה כַּאֲשֶׁר דִּמִּיתִי לַעֲשׂוֹת לָהֶם אֶעֱשֶׂה לָכֶם׃ {פ}
