	}
}
{
}

}
{
	\eProto{
		O LORD, when You came forth from Seir, Advanced from the country
		of Edom, The earth trembled; The heavens dripped, Yea, the clouds

}

	\hProto{חָדְלוּ פְרָזוֹן בְּיִשְׂרָאֵל חָדֵלּוּ עַד שַׁקַּמְתִּי דְּבוֹרָה שַׁקַּמְתִּי אֵם בְּיִשְׂרָאֵל׃}
}
{
	\eProto{

\Verse{8}

}
{
	\eProto{

}

\Verse{10}

}

	\eProto{

\Verse{11}

{
}

\Verse{12}
