\Chapter{4}

\Verse{1}
{
	\hX{וַיְהִי הַמֶּלֶךְ שְׁלֹמֹה מֶלֶךְ עַל־כּל־יִשְׂרָאֵל׃}
}
{
	\eX{King Solomon was now king over all Israel.}
}
{
}

\Verse{2}
{
	\hX{וְאֵלֶּה הַשָּׂרִים אֲשֶׁר־לוֹ עֲזַרְיָהוּ בֶן־צָדוֹק הַכֹּהֵן׃}
}
{
	\eX{These were his officials: Azariah son of Zadok—the priest;}
}
{
}

\Verse{3}
{
	\hX{אֱלִיחֹרֶף וַאֲחִיָּה בְּנֵי שִׁישָׁא סֹפְרִיםיְהוֹשָׁפָט בֶּן־אֲחִילוּד הַמַּזְכִּיר׃}
}
{
	\eX{
		Elihoreph and Ahijah sons of Shisha—scribes; Jehoshaphat son of
		Ahilud—recorder;
	}
}
{
}

\Verse{4}
{
	\hX{וּבְנָיָהוּ בֶן־יְהוֹיָדָע עַל־הַצָּבָאוְצָדוֹק וְאֶבְיָתָר כֹּהֲנִים׃}
}
{
	\eX{
		Benaiah son of Jehoiada—over the army; Zadok and Abiathar—priests;
	}
}
{
}

\Verse{5}
{
	\hX{וַעֲזַרְיָהוּ בֶן־נָתָן עַל־הַנִּצָּבִיםוְזָבוּד בֶּן־נָתָן כֹּהֵן רֵעֶה הַמֶּלֶךְ׃}
}
{
	\eX{
		Azariah son of Nathan—in charge of the prefects; Zabud son of
		Nathan the priest—companion of the king;
	}
}
{
}

\Verse{6}
{
	\hX{וַאֲחִישָׁר עַל־הַבָּיִתוַאֲדֹנִירָם בֶּן־עַבְדָּא עַל־הַמַּס׃}
}
{
	\eX{
		Ahishar—in charge of the palace; and Adoniram son of Abda—in
		charge of the forced labor.
	}
}
{
}

\Verse{7}
{
	\hX{וְלִשְׁלֹמֹה שְׁנֵים־עָשָׂר נִצָּבִים עַל־כּל־יִשְׂרָאֵל וְכִלְכְּלוּ אֶת־הַמֶּלֶךְ וְאֶת־בֵּיתוֹ חֹדֶשׁ בַּשָּׁנָה יִהְיֶה עַל־[הָאֶחָד] (אחד) לְכַלְכֵּל׃}
}
{
	\eX{
		Solomon had twelve prefects governing all Israel, who provided
		food for the king and his household; each had to provide food for
		one month in the year.
	}
}
{
	Qere Ketiv.
}

\Verse{8}
{
	\hX{וְאֵלֶּה שְׁמוֹתָם בֶּן־חוּר בְּהַר אֶפְרָיִם׃}
}
{
	\eX{
		And these were their names: Ben-hur, in the hill country of
		Ephraim;
	}
}
{
}

\Verse{9}
{
	\hX{בֶּן־דֶּקֶר בְּמָקַץ וּבְשַׁעַלְבִים וּבֵית שָׁמֶשׁ וְאֵילוֹן בֵּית חָנָן׃}
}
{
	\eX{
		Ben-deker, in Makaz, Shaalbim, Beth-shemesh, and Elon-beth-hanan;
	}
}
{
}

\Verse{10}
{
	\hX{בֶּן־חֶסֶד בָּאֲרֻבּוֹת לוֹ שֹׂכֹה וְכל־אֶרֶץ חֵפֶר׃}
}
{
	\eX{
		Ben-hesed in Arubboth—he governed Socho and all the Hepher area;
	}
}
{
}

\Verse{11}
{
	\hX{בֶּן־אֲבִינָדָב כּל־נָפַת דֹּאר טָפַת בַּת־שְׁלֹמֹה הָיְתָה לּוֹ לְאִשָּׁה׃}
}
{
	\eX{
		Ben-abinadab, [in] all of Naphath-dor (Solomon’s daughter Taphath
		was his wife);
	}
}
{
}

\Verse{12}
{
	\hX{בַּעֲנָא בֶּן־אֲחִילוּד תַּעְנַךְ וּמְגִדּוֹ וְכל־בֵּית שְׁאָן אֲשֶׁר אֵצֶל צָרְתַנָה מִתַּחַת לְיִזְרְעֶאל מִבֵּית שְׁאָן עַד אָבֵל מְחוֹלָה עַד מֵעֵבֶר לְיקְמְעָם׃}
}
{
	\eX{
		Baana son of Ahilud [in] Taanach and Megiddo and all Beth-shean,
		which is beside Zarethan, below Jezreel—from Beth-shean to
		Abel-meholah as far as the other side of Jokmeam;
	}
}
{
}

\Verse{13}
{
	\hX{בֶּן־גֶּבֶר בְּרָמֹת גִּלְעָד לוֹ חַוֺּת יָאִיר בֶּן־מְנַשֶּׁה אֲשֶׁר בַּגִּלְעָד לוֹ חֶבֶל אַרְגֹּב אֲשֶׁר בַּבָּשָׁן שִׁשִּׁים עָרִים גְּדֹלוֹת חוֹמָה וּבְרִיחַ נְחֹשֶׁת׃}
}
{
	\eX{
		Ben-geber, in Ramoth-gilead—he governed the villages of Jair son
		of Manasseh which are in Gilead, and he also governed the district
		of Argob which is in Bashan, sixty large towns with walls and
		bronze bars;
	}
}
{
}

}

	\eX{Ahinadab son of Iddo, in Mahanaim;}

{
}
