\definecolor{SpineBackground}{cmyk}{0.63,0.40,0.35,0.84}
\definecolor{SpineForeground}{cmyk}{0.00,0.20,0.62,0.30}
\newlength{\CoverFrontCenterX}
\setlength{\CoverFrontCenterX}{3.5625in}
\newlength{\CoverWeY}
% These are the centers of the two titles embedded in the front artwork after
% it is scaled and cropped below.  The volume title is typeset on the front,
% so the same three coordinates can also place the corresponding spine text.
\setlength{\CoverWeY}{5.725in}
\newlength{\CoverNamelessY}
\setlength{\CoverNamelessY}{4.425in}
\newlength{\CoverVolumeY}
\setlength{\CoverVolumeY}{1.425in}
\newcommand{\SpineLabel}[2]{%
  \adjustbox{max width=\dimexpr\CoverSpineWidth-0.125in\relax}{\fontsize{#1}{#1}\selectfont #2}%
}
\newcommand{\LD}[1][1em]{%
  \tikz[baseline=-0.15ex,x=#1,y=#1]{%
    \def\LDscale{0.82}\def\LDgap{0.16}
    \node[anchor=south west,inner sep=0pt,outer sep=0pt,scale=\LDscale]
      at (0,0) {\sffamily\bfseries L};
    \node[anchor=south west,inner sep=0pt,outer sep=0pt,rotate=180,scale=\LDscale]
      at (1+\LDgap,1+\LDgap) {\sffamily\bfseries L};
  }%
}
\def\CoverSpineLD{ld}
\newcommand{\SpineName}{%
  \ifx\CoverSpine\CoverSpineLD
    \SpineLabel{22}{\LD}%
  \else
    \SpineLabel{15}{\shortstack{The\\Nameless}}%
  \fi
}
