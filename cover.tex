% KDP paperback cover wrap. Dimensions are supplied by bin/kdp-cover from the
% finished interior PDF so a pagination change cannot silently stale the spine.
\documentclass{article}
\usepackage[
    paperwidth=\dimexpr \CoverTrimWidth*2+\CoverSpineWidth+0.25in\relax,
    paperheight=\dimexpr \CoverTrimHeight+0.25in\relax,
    margin=0pt
]{geometry}
\usepackage[cmyk]{xcolor}
\usepackage{tikz}
\usepackage{fontspec}
\pagestyle{empty}

% Print approximations of the Everforest dark title-page colors.
\definecolor{CoverBackground}{cmyk}{0.23,0.10,0.00,0.77}
\definecolor{CoverForeground}{cmyk}{0.00,0.03,0.09,0.09}

\newlength{\CoverBleed}
\setlength{\CoverBleed}{0.125in}
\newlength{\CoverFrontLeft}
\setlength{\CoverFrontLeft}{\dimexpr\CoverBleed+\CoverTrimWidth+\CoverSpineWidth\relax}
\newlength{\CoverSpineCenterX}
\setlength{\CoverSpineCenterX}{\dimexpr\CoverBleed+\CoverTrimWidth\relax}
\addtolength{\CoverSpineCenterX}{\dimexpr\CoverSpineWidth/2\relax}
\newlength{\CoverSpineCenterY}
\setlength{\CoverSpineCenterY}{\CoverBleed}
\addtolength{\CoverSpineCenterY}{\dimexpr\CoverTrimHeight/2\relax}
\newlength{\CoverFrontCenterX}
\setlength{\CoverFrontCenterX}{\dimexpr\CoverTrimWidth/2\relax}
\newlength{\CoverTitleTopY}
\setlength{\CoverTitleTopY}{\dimexpr\CoverTrimHeight*72/100\relax}
\newlength{\CoverTitleMiddleY}
\setlength{\CoverTitleMiddleY}{\dimexpr\CoverTrimHeight*46/100\relax}
\newlength{\CoverTitleBottomY}
\setlength{\CoverTitleBottomY}{\dimexpr\CoverTrimHeight*13/100\relax}

\begin{document}
\begin{tikzpicture}[remember picture,overlay,x=1in,y=1in]
    % One continuous, full-bleed background: back, spine, then front.
    \fill[CoverBackground] (current page.south west) rectangle (current page.north east);

    % Front: the same restrained hierarchy and vertical rhythm as the dark
    % standalone Genesis title page.
    \coordinate (frontSW) at ([xshift=\CoverFrontLeft,yshift=\CoverBleed]current page.south west);
    \node[CoverForeground,anchor=center] at
        ([xshift=\CoverFrontCenterX,yshift=\CoverTitleTopY]frontSW)
        {\fontsize{100}{120}\selectfont\textsc{We}};
    \node[CoverForeground,anchor=center] at
        ([xshift=\CoverFrontCenterX,yshift=\CoverTitleMiddleY]frontSW)
        {\fontsize{25}{30}\selectfont The Nameless};
    \node[CoverForeground,anchor=center] at
        ([xshift=\CoverFrontCenterX,yshift=\CoverTitleBottomY]frontSW)
        {\fontsize{23}{28}\selectfont Genesis};

    % Read bottom-to-top while upright on a shelf. The 0.0625-inch KDP fold
    % allowance remains clear on both sides of the spine text.
    \coordinate (spineCenter) at
        ([xshift=\CoverSpineCenterX,yshift=\CoverSpineCenterY]current page.south west);
    \node[CoverForeground,rotate=90,anchor=center] at (spineCenter)
        {\fontsize{16}{19}\selectfont\textsc{We}\enspace The Nameless
         \quad{\fontsize{9}{11}\selectfont Genesis}};
\end{tikzpicture}
\null
\end{document}
