    \HistoryBookActivetrue

    \def\CurrentTitleBook{history.#1}%
    \BookHeading{\TableOfContentsLink{#2}}%
    \phantomsection
    \hypertarget{book.history.#1}{}%
    \addcontentsline{toc}{section}{#2}%
}

\newcommand{\HistoryRunningTitle}{%
    \ifHistoryBookActive
        \HistoryBookTitle\HistoryLayerLabel

\let\HistoryOriginalChapterTitleText\ChapterTitleText

    \ifHistoryBookActive

}

    \hyperlink{chapter.\CurrentBook.#1}{\HistoryRunningTitle\ #1}%
}
